    <script>
        const movieForm = document.getElementById('movie-form');
        const moviesContainer = document.getElementById('movies-container');
        const movieCountEl = document.getElementById('movie-count');

        const initialMovies = [
            {
                title: "සසර සැරි සරන තෙක් ඔබ මගේ",
                meta: "Classic Drama | Vijaya Kumaratunga",
                poster: "https://images.unsplash.com/photo-1489599849927-2ee91cede3ba?q=80&w=600&auto=format&fit=crop",
                link: "#",
                desc: "සිංහල සිනමාවේ අමරණීය ආදර අමුතු කථාන්දරය සහ විශිෂ්ට රංගනයෙන් හැඩගැන්වුණු සිනමා පටයකි."
            }
        ];

        let movies = JSON.parse(localStorage.getItem('hela_flix_movies')) || initialMovies;

        function renderMovies() {
            movieCountEl.innerText = `Movies: ${movies.length}`;
            moviesContainer.innerHTML = '';

            if (movies.length === 0) {
                moviesContainer.innerHTML = `<p class="col-span-full text-center text-slate-500 py-12">තවම චිත්‍රපට එකතු කර නැත.</p>`;
                return;
            }

            movies.forEach((movie) => {
                const card = document.createElement('div');
                card.className = "bg-slate-900 border border-slate-800 rounded-xl overflow-hidden shadow-xl hover:border-amber-500/50 transition duration-300 flex flex-col";
                card.innerHTML = `
                    <div class="h-56 overflow-hidden relative">
                        <img src="${escapeHtml(movie.poster)}" alt="${escapeHtml(movie.title)}" class="w-full h-full object-cover hover:scale-105 transition duration-500">
                        <div class="absolute bottom-0 left-0 right-0 bg-gradient-to-t from-slate-900 via-slate-900/60 to-transparent p-4">
                            <span class="text-xs font-semibold text-amber-400 bg-amber-500/20 px-2 py-0.5 rounded border border-amber-500/30">${escapeHtml(movie.meta)}</span>
                        </div>
                    </div>
                    <div class="p-5 flex-1 flex flex-col justify-between space-y-4">
                        <div>
                            <h4 class="text-lg font-bold text-white mb-2">${escapeHtml(movie.title)}</h4>
                            <p class="text-xs text-slate-400 line-clamp-3 leading-relaxed">${escapeHtml(movie.desc)}</p>
                        </div>
                        ${movie.link ? `
                            <a href="${escapeHtml(movie.link)}" target="_blank" class="block text-center text-xs font-bold text-amber-400 bg-slate-950 border border-amber-500/40 py-2.5 rounded-lg hover:bg-amber-500 hover:text-slate-950 transition">
                                Watch Trailer / Details ➔
                            </a>
                        ` : ''}
                    </div>
                `;
                moviesContainer.appendChild(card);
            });
        }

        movieForm.addEventListener('submit', (e) => {
            e.preventDefault();
            const newMovie = {
                title: document.getElementById('movie-title').value,
                meta: document.getElementById('movie-meta').value,
                poster: document.getElementById('movie-poster').value,
                link: document.getElementById('movie-link').value,
                desc: document.getElementById('movie-desc').value
            };

            movies.unshift(newMovie);
            localStorage.setItem('hela_flix_movies', JSON.stringify(movies));
            movieForm.reset();
            renderMovies();
        });

        function escapeHtml(text) {
            return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
        }

        renderMovies();
    </script>
